#include <iostream>

int my_function() {
    std::cout << "Hello, world!" << std::endl;
    return 0;
}

int main() {
    // Original instructions
    int x = 0x12345678;
    int *ptr = &x;
    *ptr = 0x87654321;

    // Obfuscated instructions using pointers
    int y = 0x12345678;
    int *ptr1 = &y;
    int *ptr2 = ptr1;
    *ptr2 = 0x87654321;

    return 0;
}
